  <script>
    // ----- CONFIG -----
    // Set these to your real image width/height in pixels
    const IMAGE_WIDTH = 1743;   // approximate from your screenshot
    const IMAGE_HEIGHT = 768;

    // Marker list: x & y in image pixels
    // Add/remove/edit these as you like
    const MARKERS = [
      {
        id: "unionCity",
        name: "UNION CITY",
        x: 1350,
        y: 260,
        description:
"UNION CITY\nPrimary seat of the Midwestern Provisional Authority.\nHigh security. Access by clearance only."
      },
      {
        id: "crossroads",
        name: "CROSSROADS STATION",
        x: 1150,
        y: 420,
        description:
"CROSSROADS STATION\nFrontier trading outpost on the western approach to Chicago.\nCaravans, mercenaries, smugglers."
      },
      {
        id: "littleItaly",
        name: "LITTLE ITALY",
        x: 1320,
        y: 310,
        description:
"LITTLE ITALY – THE CATACOMBS\nTerritory of Alessandro 'Old Capone' Martelli\nand the multi-family Outfit syndicate."
      },
      {
        id: "raiderWest",
        name: "RAIDER FRONTIER",
        x: 900,
        y: 400,
        description:
"UNIFIED RAIDER TERRITORY\nCoordinated strikes reported.\nTravel not recommended."
      },
      {
        id: "mutantSouth",
        name: "MUTANT BELT",
        x: 1320,
        y: 520,
        description:
"INDUSTRIAL MUTANT BELT\nHigh FEV contamination.\nMutant warbands active."
      }
    ];

    // ----- STATE -----
    const mapViewport = document.getElementById("mapViewport");
    const mapInner = document.getElementById("mapInner");
    const mapImage = document.getElementById("mapImage");
    const infoPanel = document.getElementById("infoPanel");
    const coordsEl = document.getElementById("coords");

    const zoomInBtn = document.getElementById("zoomInBtn");
    const zoomOutBtn = document.getElementById("zoomOutBtn");
    const resetBtn = document.getElementById("resetBtn");

    // pan/zoom state
    let scale = 1;
    let offsetX = 0;
    let offsetY = 0;

    let isDragging = false;
    let dragStartX = 0;
    let dragStartY = 0;
    let dragOriginX = 0;
    let dragOriginY = 0;

    function updateTransform() {
      mapInner.style.transform =
        `translate(${offsetX}px, ${offsetY}px) scale(${scale})`;
      coordsEl.textContent =
        `X: ${Math.round(-offsetX / scale)}, ` +
        `Y: ${Math.round(-offsetY / scale)}, ` +
        `ZOOM: ${scale.toFixed(2)}`;
    }

    function clamp(val, min, max) {
      return Math.max(min, Math.min(max, val));
    }

    function setScale(newScale, centerX, centerY) {
      const oldScale = scale;
      newScale = clamp(newScale, 0.5, 3.5);

      if (newScale === oldScale) return;

      // Zoom towards cursor point
      const rect = mapViewport.getBoundingClientRect();
      const cx = centerX - rect.left;
      const cy = centerY - rect.top;

      const wx = (cx - offsetX) / oldScale;
      const wy = (cy - offsetY) / oldScale;

      scale = newScale;
      offsetX = cx - wx * scale;
      offsetY = cy - wy * scale;
      updateTransform();
    }

    // ----- MARKERS -----
    function createMarkers() {
      MARKERS.forEach(m => {
        const el = document.createElement("div");
        el.className = "marker";
        el.dataset.id = m.id;

        const label = document.createElement("div");
        label.className = "marker-label";
        label.textContent = m.name;
        el.appendChild(label);

        // Position in image coordinates; we convert in CSS transform space
        const xRatio = m.x / IMAGE_WIDTH;
        const yRatio = m.y / IMAGE_HEIGHT;
        el.style.left = `${xRatio * 100}%`;
        el.style.top = `${yRatio * 100}%`;

        el.addEventListener("click", (e) => {
          e.stopPropagation();
          selectMarker(m);
        });

        mapInner.appendChild(el);
      });
    }

    function selectMarker(marker) {
      infoPanel.textContent = marker.description;
    }

    // ----- EVENTS -----
    mapViewport.addEventListener("mousedown", (e) => {
      isDragging = true;
      mapViewport.style.cursor = "grabbing";
      dragStartX = e.clientX;
      dragStartY = e.clientY;
      dragOriginX = offsetX;
      dragOriginY = offsetY;
    });

    window.addEventListener("mouseup", () => {
      isDragging = false;
      mapViewport.style.cursor = "grab";
    });

    window.addEventListener("mousemove", (e) => {
      if (!isDragging) return;
      const dx = e.clientX - dragStartX;
      const dy = e.clientY - dragStartY;
      offsetX = dragOriginX + dx;
      offsetY = dragOriginY + dy;
      updateTransform();
    });

    mapViewport.addEventListener("wheel", (e) => {
      e.preventDefault();
      const delta = e.deltaY < 0 ? 0.1 : -0.1;
      setScale(scale + delta, e.clientX, e.clientY);
    }, { passive: false });

    zoomInBtn.addEventListener("click", () => {
      const rect = mapViewport.getBoundingClientRect();
      setScale(scale + 0.25, rect.left + rect.width / 2, rect.top + rect.height / 2);
    });

    zoomOutBtn.addEventListener("click", () => {
      const rect = mapViewport.getBoundingClientRect();
      setScale(scale - 0.25, rect.left + rect.width / 2, rect.top + rect.height / 2);
    });

    resetBtn.addEventListener("click", () => {
      scale = 1;
      offsetX = 0;
      offsetY = 0;
      updateTransform();
      infoPanel.textContent =
"MPA REGIONAL NAVIGATION ONLINE.\nCLICK A MARKER FOR INTEL.";
    });

    // Click empty space to clear info
    mapViewport.addEventListener("click", () => {
      infoPanel.textContent =
"NO ACTIVE SELECTION.\nUse markers or controls to navigate.";
    });

    // ----- INIT -----
    createMarkers();
    updateTransform();
  </script>
